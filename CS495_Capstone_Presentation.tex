        \item \textbf{Cox, Ross \& Rubinstein (1979):} binomial
              lattice; no ML integration
        \item \textbf{Black \& Scholes (1973):} closed-form; fails
              for American early exercise
        \item \textbf{Bakshi et al.\ (1997):} stochastic vol; fit
              requires unobservable parameters
        \item \textbf{Carr \& Wu (2009):} IV overstates realized vol
              (variance risk premium)
        \item \textbf{Kelly (1956):} log-utility bet sizing; applied
              to markets by Thorp (1969)
        \item \textbf{Gap:} No framework cross-validates CRR + ML
              with regime-conditioned Kelly sizing
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.05em}

  \descpanelflex
    {What this pass does}
    {Lays out every possible stock price path from $t=0$ to expiry.
     $u$, $d$, $p^*$ derived from $\sigma = \text{RV30}$. No option
     value is computed here --- only $S$ evolution. 5-step
     visualization; 50-step lattice used for pricing.}
    {No-arbitrage}
    {No risk-free profit can exist in an efficient market. The
     risk-neutral probability $p^* = (e^{r\,\Delta t} - d)/(u-d)$
     enforces this so the discounted expected next-step stock price
     equals today's price.}
    {\emph{\S 2.2 (CRR foundational equations)} and
     \emph{\S 3.4.1 (forward-pass lattice construction)} of the
     research report.}
\end{frame}
